/* ========================================================= حفظ الحضور في Google Sheets ========================================================= */ let attendanceSaving = false; async function saveAttendanceToSheets(){ if(attendanceSaving){ return; } const section = document .getElementById("classSelect") .value; const date = document .getElementById("attendanceDate") .value; const students = Object.values(attendanceStudents) .map(student => ({ id: student.id, code: student.code, name: student.name, status: student.status || "حاضر" })); if(!section){ alert("اختر الشعبة أولاً"); return; } if(!date){ alert("اختر تاريخ الحضور"); return; } if(students.length === 0){ alert("لم يتم تسجيل أي طالب حاضر حتى الآن"); return; } const confirmed = confirm( "سيتم حفظ حضور الشعبة " + section + "\n\nعدد الحاضرين: " + students.length + "\nعدد الغائبين: " + ( classStudents[section].length - students.length ) + "\n\nهل تريد الحفظ؟" ); if(!confirmed){ return; } attendanceSaving = true; const button = document .getElementById("saveAttendanceBtn"); const status = document .getElementById("status"); button.disabled = true; button.innerHTML = "⏳ جاري الحفظ..."; status.innerHTML = "⏳ جاري إرسال الحضور إلى Google Sheets..."; const payload = { type: "attendance", section: section, date: date, students: students }; try{ await fetch( WEB_APP_URL, { method: "POST", mode: "no-cors", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify(payload) } ); status.innerHTML = "✅ تم إرسال الحضور — تحقق من ورقة الحضور"; alert( "تم إرسال حضور الشعبة " + section + "\n\nالحاضرون: " + students.length + "\n\nافتح ورقة «الحضور» الآن للتأكد من وصول البيانات." ); } catch(error){ console.error(error); status.innerHTML = "❌ تعذر إرسال الحضور"; alert( "حدث خطأ أثناء إرسال الحضور." ); } finally{ attendanceSaving = false; button.disabled = false; button.innerHTML = "💾 حفظ الحضور"; } }
